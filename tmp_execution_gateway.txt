def _utc_now() -> str:
    return datetime.now(timezone.utc).isoformat()


def _json_fingerprint(value: Any) -> str:
    encoded = json.dumps(value, default=str, sort_keys=True, separators=(",", ":"))
    return hashlib.sha256(encoded.encode("utf-8")).hexdigest()


def _norm_path(path: Any) -> str:
    text = str(path or "").replace("\\", "/").strip()
    while text.startswith("./"):
        text = text[2:]
    return text.rstrip("/")


def _normalize_path_list(value: Any) -> tuple[str, ...]:
    if value is None:
        return ()
    if isinstance(value, str):
        values = [value]
    elif isinstance(value, Mapping):
        values = list(value.values())
    else:
        try:
            values = list(value)
        except TypeError:
            values = [value]
    return tuple(dict.fromkeys(_norm_path(item) for item in values if _norm_path(item)))


def _is_under(path: str, root: str) -> bool:
    path = _norm_path(path)
    root = _norm_path(root)
    if not root:
        return True
    return path == root or path.startswith(root + "/")


def _extract_target_paths_from_mapping(value: Mapping[str, Any] | None) -> tuple[str, ...]:
    if not isinstance(value, Mapping):
        return ()

    paths: list[str] = []

    for key in ("target_file", "target_path", "path", "file"):
        item = value.get(key)
        if item:
            paths.extend(_normalize_path_list(item))

    for key in ("target_files", "target_paths", "affected_files", "impacted_files", "allow_target_paths"):
        item = value.get(key)
        if item:
            paths.extend(_normalize_path_list(item))

    edit_plan = value.get("edit_plan")
    if isinstance(edit_plan, Mapping):
        for key in ("target_files", "impacted_files", "affected_files"):
            item = edit_plan.get(key)
            if item:
                paths.extend(_normalize_path_list(item))
        for action in edit_plan.get("actions") or []:
            if isinstance(action, Mapping):
                paths.extend(_extract_target_paths_from_mapping(action))

    runtime_args = value.get("runtime_args")
    if isinstance(runtime_args, Mapping):
        paths.extend(_extract_target_paths_from_mapping(runtime_args))

    payload = value.get("payload")
    if isinstance(payload, Mapping):
        paths.extend(_extract_target_paths_from_mapping(payload))

    return tuple(dict.fromkeys(path for path in paths if path))


def _extract_target_paths_from_request(request: RuntimeExecutionRequest) -> tuple[str, ...]:
    paths: list[str] = []
    paths.extend(_extract_target_paths_from_mapping(request.metadata))
    paths.extend(_extract_target_paths_from_mapping(request.lineage))

    if isinstance(request.command, (list, tuple)):
        for item in request.command:
            text = _norm_path(item)
            if text.startswith(("core/", "tests/", "scripts/", "workspace/")):
                paths.append(text)

    return tuple(dict.fromkeys(path for path in paths if path))


def _resolve_allow_paths(metadata: Mapping[str, Any] | None, lineage: Mapping[str, Any] | None = None) -> tuple[str, ...]:
    values: list[str] = []
    for source in (metadata, lineage):
        if not isinstance(source, Mapping):
            continue
        for key in ("allow_paths", "allowed_paths", "execution_allow_paths", "bounded_allow_paths"):
            if source.get(key):
                values.extend(_normalize_path_list(source.get(key)))
        boundary = source.get("execution_boundary")
        if isinstance(boundary, Mapping):
            for key in ("allow_paths", "allowed_paths"):
                if boundary.get(key):
                    values.extend(_normalize_path_list(boundary.get(key)))
    return tuple(dict.fromkeys(values))


def _validate_allow_path_boundary(target_paths: Iterable[str], allow_paths: Iterable[str]) -> dict[str, Any]:
    targets = tuple(_normalize_path_list(target_paths))
    allowed = tuple(_normalize_path_list(allow_paths))

    if not allowed:
        return {"ok": True, "reason": "no_allow_paths_boundary", "target_paths": list(targets), "allow_paths": [], "blocked_paths": []}

    blocked = [path for path in targets if not any(_is_under(path, root) for root in allowed)]
    return {
        "ok": not blocked,
        "reason": "allow_path_boundary_ok" if not blocked else "target_path_outside_allow_paths",
        "target_paths": list(targets),
        "allow_paths": list(allowed),
        "blocked_paths": blocked,
    }


def _normalize_command(command: str | Sequence[str]) -> str | tuple[str, ...]:
    if isinstance(command, str):
        return command
    return tuple(str(item) for item in command)


def _stable_id(prefix: str, value: Any) -> str:
    digest = hashlib.sha256(repr(value).encode("utf-8", errors="replace")).hexdigest()[:16]
    return f"{prefix}:{digest}"


def _build_runtime_identity() -> dict[str, Any]:
    return {
        "identity_id": "system:runtime_execution_gateway",
        "identity_type": "SYSTEM",
        "source": RUNTIME_AUTHORITY_SOURCE,
        "canonical_owner": RUNTIME_AUTHORITY_OWNER,
    }

    shell: bool,

    text: bool,
    encoding: str,
    errors: str,
    extra_metadata: Mapping[str, Any] | None = None,
    allow_paths: Any = None,

    transaction_boundary = build_transaction_boundary_metadata(
        {
            "transaction_source": "runtime_execution_gateway",
            "transaction_status": "opened",
            "transaction_scope": "execution_gateway",
        }
    )
    normalized_allow_paths = _normalize_path_list(allow_paths)
    metadata: dict[str, Any] = {
        "task_id": "runtime_execution_gateway",
        "step_id": _stable_id("gateway_step", command),
        "runtime_session": _stable_id("runtime_session", time.time_ns()),
        "approval_state": "approved",
        "policy_result": {"allowed": True, "source": RUNTIME_AUTHORITY_SOURCE},
        "trace_id": _stable_id("trace", command),
        "source": "runtime_execution_gateway",
        "execution_source": "runtime_execution_gateway",
        "gateway": "canonical_execution_gateway",
        "runtime_authority_entrypoint": RUNTIME_AUTHORITY_ENTRYPOINT,
        "runtime_authority_source": RUNTIME_AUTHORITY_SOURCE,
        "authority_source": RUNTIME_AUTHORITY_SOURCE,
        "authority_scope": RUNTIME_AUTHORITY_SCOPE_ID,
        "authority_status": "allowed",
        "authority_reason": "runtime_execution_gateway_authorized",
        "ownership_source": RUNTIME_AUTHORITY_OWNER,
        "ownership_scope": RUNTIME_CAPABILITY_SCOPE_ID,
        "canonical_owner": RUNTIME_AUTHORITY_OWNER,
        "execution_authority_unified": True,
        "execution_legality": "legal",
        "direct_subprocess_bypass": False,
        "shell": shell,
        "input_text_ignored": input_text is not None,
        "capture_output": capture_output,
        "text": text,
        "encoding": encoding,
        "errors": errors,
        "runtime_identity": _build_runtime_identity(),
        "authority_scope_id": RUNTIME_AUTHORITY_SCOPE_ID,
        "capability_scope_id": RUNTIME_CAPABILITY_SCOPE_ID,
        "transaction_boundary": transaction_boundary,
        "execution_boundary": {
            "boundary_type": "allow_paths",
            "allow_paths": list(normalized_allow_paths),
            "enforced_by": RUNTIME_AUTHORITY_SOURCE,
        },
        "provenance": {
            "requested_by": "core.runtime.execution_gateway.safe_subprocess_run",
            "gateway": "canonical_execution_gateway",
            "canonical_owner": RUNTIME_AUTHORITY_OWNER,
            "bypass_prevention": "subprocess execution is delegated to Executor.execute_request",
        },
    }

    if normalized_allow_paths:
        metadata["allow_paths"] = list(normalized_allow_paths)
        metadata["allowed_paths"] = list(normalized_allow_paths)
        metadata["execution_allow_paths"] = list(normalized_allow_paths)

    if extra_metadata:
        metadata.update(dict(extra_metadata))
        merged_allow = _resolve_allow_paths(metadata)
        metadata["execution_boundary"] = {
            "boundary_type": "allow_paths",
            "allow_paths": list(merged_allow),
            "enforced_by": RUNTIME_AUTHORITY_SOURCE,
        }
        if merged_allow:
            metadata["allow_paths"] = list(merged_allow)
            metadata["allowed_paths"] = list(merged_allow)
            metadata["execution_allow_paths"] = list(merged_allow)

    closure = build_runtime_closure_fields(
        {**metadata, "execution_status": metadata.get("execution_status", "opened"), "source": "runtime_execution_gateway"},
        artifact_type="execution_gateway",
        artifact_id=RUNTIME_AUTHORITY_ENTRYPOINT,
        finalized_by="runtime_execution_gateway",
    )
    metadata.update(closure)
    metadata["consistency_seal"] = build_runtime_state_consistency(metadata)
    metadata["authority_validation"] = {"ok": True, "reason": "authority_metadata_valid", "missing_fields": []}
    return metadata


def _extract_existing_rollback_verification(metadata: Mapping[str, Any]) -> dict[str, Any]:
    for key in ("rollback_verification", "rollback_verification_result", "rollback_gate", "rollback_gate_result"):
        value = metadata.get(key)
        if isinstance(value, Mapping):
            return copy.deepcopy(dict(value))
    return {"ok": None, "attached": False, "reason": "rollback_verification_not_attached"}


def _build_replay_verification(*, replay_id: str | None, metadata: Mapping[str, Any], ok: bool) -> dict[str, Any]:
    replay_event = metadata.get("replay_event")
    replay_chain = metadata.get("replay_chain")
    replay_fingerprint = metadata.get("replay_fingerprint")
    attached = bool(replay_id or replay_event or replay_chain or replay_fingerprint)
    return {
        "ok": bool(ok) if attached else None,
        "attached": attached,
        "replay_id": replay_id,
        "replay_event": copy.deepcopy(replay_event),
        "replay_chain": copy.deepcopy(replay_chain),
        "replay_fingerprint": replay_fingerprint,
        "reason": "replay_verification_attached" if attached else "replay_verification_not_attached",
    }


def _extract_existing_outcome_verification(metadata: Mapping[str, Any]) -> dict[str, Any] | None:
    for key in ("outcome_verification", "rollback_outcome_verification", "execution_outcome_verification"):
        value = metadata.get(key)
        if isinstance(value, Mapping):
            return copy.deepcopy(dict(value))
    gate = metadata.get("rollback_gate_result")
    if isinstance(gate, Mapping):
        value = gate.get("outcome_verification")
        if isinstance(value, Mapping):
            return copy.deepcopy(dict(value))
    return None


def _build_gateway_outcome_verification(
    *,
    ok: bool,
    rollback_verification: Mapping[str, Any],
    replay_verification: Mapping[str, Any],
    existing_outcome_verification: Mapping[str, Any] | None,
    metadata: Mapping[str, Any],
) -> ExecutionGatewayOutcomeVerification:
    created_at = _utc_now()

    if existing_outcome_verification:
        verification_state = str(existing_outcome_verification.get("verification_state") or "")
        deterministic = bool(existing_outcome_verification.get("deterministic", verification_state == "verified"))
        rollback_verified = bool(existing_outcome_verification.get("rollback_verified", True))
        replay_verified = bool(existing_outcome_verification.get("replay_verified", replay_verification.get("attached")))
        legality_verified = bool(existing_outcome_verification.get("legality_verified", True))
        frozen = bool(existing_outcome_verification.get("frozen", verification_state != "verified"))
        reason = str(existing_outcome_verification.get("reason") or "gateway_outcome_verification_attached")
    else:
        rollback_attached = bool(rollback_verification.get("attached", True))
        rollback_ok = rollback_verification.get("ok")
        replay_attached = bool(replay_verification.get("attached"))
        replay_ok = replay_verification.get("ok")

        rollback_verified = rollback_ok is True or rollback_verification.get("verification_result") == "verified" or rollback_verification.get("verification_state") == "verified"
        replay_verified = (not replay_attached) or replay_ok is True
        legality_verified = bool(metadata.get("execution_legality", "legal") == "legal")
        deterministic = bool(ok and rollback_verified and replay_verified and legality_verified)
        verification_state = "verified" if deterministic else "mismatched"
        frozen = verification_state != "verified"
        reason = "gateway_outcome_verification_verified" if deterministic else "gateway_outcome_verification_mismatched"

    payload = {
        "ok": ok,
        "rollback_verification": rollback_verification,
        "replay_verification": replay_verification,
        "existing_outcome_verification": existing_outcome_verification,
        "verification_state": verification_state,
        "deterministic": deterministic,
        "rollback_verified": rollback_verified,
        "replay_verified": replay_verified,
        "legality_verified": legality_verified,
        "frozen": frozen,
        "reason": reason,
        "created_at": created_at,
    }
    verification_fingerprint = _json_fingerprint(payload)

    return ExecutionGatewayOutcomeVerification(
        verification_id=f"execution_gateway_outcome_verification:{verification_fingerprint[:16]}",
        verification_state=verification_state,
        deterministic=deterministic,
        replay_verified=replay_verified,
        rollback_verified=rollback_verified,
        legality_verified=legality_verified,
        frozen=frozen,
        reason=reason,
        verification_fingerprint=verification_fingerprint,
        created_at=created_at,
    )


def _build_gateway_outcome_closure(
    *,
    ok: bool,
    returncode: int | None,
    stdout: str,
    stderr: str,
    command: Any,
    shell: bool,
    timeout: float | None,
    error: str | None,
    replay_id: str | None,
    metadata: Mapping[str, Any],
    risk_metadata: Mapping[str, Any],
) -> ExecutionGatewayOutcomeClosure:
    outcome_state = "succeeded" if ok else "failed"
    reason = "execution_gateway_outcome_succeeded" if ok else str(error or stderr or "execution_gateway_outcome_failed")

    rollback_verification = _extract_existing_rollback_verification(metadata)
    replay_verification = _build_replay_verification(replay_id=replay_id, metadata=metadata, ok=ok)
    existing_outcome_verification = _extract_existing_outcome_verification(metadata)

    outcome_verification = _build_gateway_outcome_verification(
        ok=ok,
        rollback_verification=rollback_verification,
        replay_verification=replay_verification,
        existing_outcome_verification=existing_outcome_verification,
        metadata=metadata,
    )

    frozen = (not ok) or outcome_verification.frozen

    command_payload = {
        "command": command,
        "shell": shell,
        "timeout": timeout,
        "returncode": returncode,
        "stdout": stdout,
        "stderr": stderr,
        "error": error,
    }

    command_fingerprint = _json_fingerprint(command_payload)
    metadata_fingerprint = _json_fingerprint(metadata)
    risk_fingerprint = _json_fingerprint(risk_metadata)

    created_at = _utc_now()
    outcome_id = _stable_id(
        "execution_gateway_outcome",
        {
            "command_fingerprint": command_fingerprint,
            "metadata_fingerprint": metadata_fingerprint,
            "risk_fingerprint": risk_fingerprint,
            "outcome_verification": outcome_verification.to_dict(),
            "created_at": created_at,
        },
    )

    outcome_payload = {
        "outcome_id": outcome_id,
        "outcome_state": outcome_state,
        "finalized": True,
        "frozen": frozen,
        "reason": reason,
        "command_fingerprint": command_fingerprint,
        "metadata_fingerprint": metadata_fingerprint,
        "risk_fingerprint": risk_fingerprint,
        "replay_id": replay_id,
        "rollback_verification": rollback_verification,
        "replay_verification": replay_verification,
        "outcome_verification": outcome_verification.to_dict(),
        "created_at": created_at,
    }

    return ExecutionGatewayOutcomeClosure(
        outcome_id=outcome_id,
        outcome_state=outcome_state,
        finalized=True,
        frozen=frozen,
        reason=reason,
        command_fingerprint=command_fingerprint,
        metadata_fingerprint=metadata_fingerprint,
        risk_fingerprint=risk_fingerprint,
        outcome_fingerprint=_json_fingerprint(outcome_payload),
        replay_id=replay_id,
        rollback_verification=rollback_verification,
        replay_verification=replay_verification,
        outcome_verification=outcome_verification.to_dict(),
        created_at=created_at,
    )


def _attach_outcome_metadata(
    *,
    metadata: Mapping[str, Any],
    risk_metadata: Mapping[str, Any],
    outcome: ExecutionGatewayOutcomeClosure,
) -> tuple[dict[str, Any], dict[str, Any]]:
    outcome_dict = outcome.to_dict()
    verification = dict(outcome.outcome_verification)

    closed_metadata = {
        **dict(metadata),
        "execution_outcome_closure": outcome_dict,
        "execution_outcome_id": outcome.outcome_id,
        "execution_outcome_state": outcome.outcome_state,
        "execution_outcome_finalized": outcome.finalized,
        "execution_outcome_frozen": outcome.frozen,
        "execution_outcome_fingerprint": outcome.outcome_fingerprint,
        "execution_outcome_verification": verification,
        "execution_outcome_verification_state": verification.get("verification_state"),
        "execution_outcome_verification_fingerprint": verification.get("verification_fingerprint"),
        "execution_status": "finalized",
        "finalized": True,
        "finalized_by": RUNTIME_AUTHORITY_SOURCE,
        "finalized_at": outcome.created_at,
        "runtime_frozen": outcome.frozen,
        "freeze_reason": outcome.reason if outcome.frozen else None,
    }

    closed_metadata["consistency_seal"] = build_runtime_state_consistency(closed_metadata)

    closed_risk_metadata = {
        **dict(risk_metadata),
        "execution_outcome_closure": outcome_dict,
        "execution_outcome_fingerprint": outcome.outcome_fingerprint,
        "execution_outcome_verification": verification,
        "runtime_frozen": outcome.frozen,
        "freeze_reason": outcome.reason if outcome.frozen else None,
    }

    return closed_metadata, closed_risk_metadata


def _finalize_gateway_result(
    *,
    ok: bool,
    returncode: int | None,
    stdout: str,
    stderr: str,
    command: Any,
    shell: bool,
    timeout: float | None,
    error: str | None,
    replay_id: str | None,
    metadata: Mapping[str, Any],
    risk_metadata: Mapping[str, Any],
) -> ExecutionGatewayResult:
    outcome = _build_gateway_outcome_closure(
        ok=ok,
        returncode=returncode,
        stdout=stdout,
        stderr=stderr,
        command=command,
        shell=shell,
        timeout=timeout,
        error=error,
        replay_id=replay_id,
        metadata=metadata,
        risk_metadata=risk_metadata,
    )
    closed_metadata, closed_risk_metadata = _attach_outcome_metadata(metadata=metadata, risk_metadata=risk_metadata, outcome=outcome)
    return ExecutionGatewayResult(
        ok=ok,
        returncode=returncode,
        stdout=stdout,
        stderr=stderr,
        command=command,
        shell=shell,
        timeout=timeout,
        error=error,
        replay_id=replay_id,
        metadata=closed_metadata,
        risk_metadata=closed_risk_metadata,
        outcome_closure=outcome.to_dict(),
    )


def build_runtime_execution_request(
    command: str | Sequence[str],
    *,
    shell: bool = False,
    cwd: str | None = None,
    timeout: float | None = 60.0,
    input_text: str | None = None,
    env: dict[str, str] | None = None,
    capture_output: bool = True,
    text: bool = True,
    encoding: str = "utf-8",
    errors: str = "replace",
    metadata: Mapping[str, Any] | None = None,
    lineage: Mapping[str, Any] | None = None,
    operator_session_id: str | None = None,
    context: Mapping[str, Any] | None = None,
    allow_paths: Any = None,
) -> RuntimeExecutionRequest:
    normalized_command = _normalize_command(command)
    surface = classify_runtime_surface("command" if shell else "subprocess")
    forwarded_metadata: dict[str, Any] = dict(metadata or {})
    forwarded_lineage: dict[str, Any] = dict(lineage or {})
    context_session_id = ""
    context_allow_paths: Any = None

    if isinstance(context, Mapping):
        context_session_id = str(context.get("operator_session_id") or context.get("persistent_operator_session_id") or "").strip()
        context_allow_paths = context.get("allow_paths") or context.get("allowed_paths") or context.get("execution_allow_paths")

    resolved_operator_session_id = str(operator_session_id or context_session_id or "").strip()
    if resolved_operator_session_id:
        forwarded_metadata.setdefault("operator_session_id", resolved_operator_session_id)
        forwarded_lineage.setdefault("operator_session_id", resolved_operator_session_id)

    normalized_allow_paths = _normalize_path_list(allow_paths) or _normalize_path_list(context_allow_paths) or _resolve_allow_paths(forwarded_metadata, forwarded_lineage)

    if normalized_allow_paths:
        forwarded_metadata.setdefault("allow_paths", list(normalized_allow_paths))
        forwarded_metadata.setdefault("allowed_paths", list(normalized_allow_paths))
        forwarded_metadata.setdefault("execution_allow_paths", list(normalized_allow_paths))
        forwarded_lineage.setdefault("allow_paths", list(normalized_allow_paths))

    request_metadata = _build_authority_metadata(
        command=normalized_command,
        shell=shell,
        input_text=input_text,
        capture_output=capture_output,
        text=text,
        encoding=encoding,
        errors=errors,
        extra_metadata=forwarded_metadata,
        allow_paths=normalized_allow_paths,
    )

    request_lineage: dict[str, Any] = {
        "execution_start_id": "execution_start:runtime_execution_gateway",
        "authority_entrypoint": RUNTIME_AUTHORITY_ENTRYPOINT,
        "authority_source": RUNTIME_AUTHORITY_SOURCE,
        "canonical_owner": RUNTIME_AUTHORITY_OWNER,
    }

    if forwarded_lineage:
        request_lineage.update(forwarded_lineage)

    return RuntimeExecutionRequest(
        execution_type="command" if shell else "subprocess",
        command=normalized_command,
        working_directory=cwd,
        environment=env,
        timeout=timeout,
        metadata={**request_metadata, "runtime_surface": surface.name},
        lineage=request_lineage,
    )


def execute_runtime_request(
    request: RuntimeExecutionRequest,
    *,
    workspace_root: str = "workspace",
    executor: Executor | None = None,
) -> ExecutionGatewayResult:
    normalized_metadata, authority_validation = ensure_authority_metadata(
        request.metadata,
        lineage=request.lineage,
        authority_source=str(request.metadata.get("authority_source") or "execution_gateway"),
        action_type="execute",
        surface=request.execution_type,
    )

    if normalized_metadata != request.metadata:
        request = RuntimeExecutionRequest(
            execution_type=request.execution_type,
            command=request.command,
            working_directory=request.working_directory,
            environment=request.environment,
            timeout=request.timeout,
            metadata=normalized_metadata,
            lineage=dict(request.lineage),
            replay_id=request.replay_id,
            repair_session_id=request.repair_session_id,
            dry_run=request.dry_run,
        )

    if not authority_validation.get("ok"):
        reason = str(authority_validation.get("reason") or "authority_metadata_invalid")
        metadata = {
            **dict(request.metadata),
            "authority_validation": authority_validation,
            "blocked": True,
            "blocked_reason": reason,
            "audit_event": {"event_type": "execution_blocked", "reason": reason, "source": RUNTIME_AUTHORITY_SOURCE},
            "evidence": {"blocked": True, "reason": reason, "authority_validation": authority_validation},
            "replay_event": {"event_type": "execution_decision", "decision": "blocked", "reason": reason},
        }
        return _finalize_gateway_result(
            ok=False,
            returncode=1,
            stdout="",
            stderr=reason,
            command=request.command,
            shell=request.execution_type == "command",
            timeout=request.timeout,
            error=reason,
            replay_id=request.replay_id,
            metadata=metadata,
            risk_metadata={"authority_validation": authority_validation},
        )

    allow_paths = _resolve_allow_paths(request.metadata, request.lineage)
    target_paths = _extract_target_paths_from_request(request)
    boundary_validation = _validate_allow_path_boundary(target_paths, allow_paths)

    if not boundary_validation["ok"]:
        reason = str(boundary_validation["reason"])
        metadata = {
            **dict(request.metadata),
            "authority_validation": authority_validation,
            "execution_boundary_validation": boundary_validation,
            "blocked": True,
            "blocked_reason": reason,
            "audit_event": {"event_type": "execution_blocked", "reason": reason, "source": RUNTIME_AUTHORITY_SOURCE, "boundary": boundary_validation},
            "evidence": {"blocked": True, "reason": reason, "boundary": boundary_validation},
            "replay_event": {"event_type": "execution_decision", "decision": "blocked", "reason": reason, "boundary": boundary_validation},
        }
        return _finalize_gateway_result(
            ok=False,
            returncode=1,
            stdout="",
            stderr=reason,
            command=request.command,
            shell=request.execution_type == "command",
            timeout=request.timeout,
            error=reason,
            replay_id=request.replay_id,
            metadata=metadata,
            risk_metadata={"authority_validation": authority_validation, "execution_boundary_validation": boundary_validation},
        )

    runtime_executor = executor if executor is not None else Executor(workspace_root=workspace_root)
    result = runtime_executor.execute_request(request)

    stderr_text = str(result.stderr or "")
    timed_out = result.return_code == 124 and "timeout" in stderr_text.lower()
    error = f"timeout after {request.timeout} seconds" if timed_out else None
    returncode = None if timed_out else result.return_code

    metadata = {
        **dict(result.metadata),
        "authority_validation": authority_validation,
        "execution_boundary_validation": boundary_validation,
    }
    risk_metadata = {
        **dict(result.risk_metadata),
        "authority_validation": authority_validation,
        "execution_boundary_validation": boundary_validation,
    }

    return _finalize_gateway_result(
        ok=result.return_code == 0,
        returncode=returncode,
        stdout=result.stdout,
        stderr=result.stderr,
        command=request.command,
        shell=request.execution_type == "command",
        timeout=request.timeout,
        error=error,
        replay_id=result.replay_id,
        metadata=metadata,
        risk_metadata=risk_metadata,
    )


def safe_subprocess_run(
    command: str | Sequence[str],
    *,
    shell: bool = False,
    cwd: str | None = None,
    timeout: float | None = 60.0,
    input_text: str | None = None,
    env: dict[str, str] | None = None,
    capture_output: bool = True,
    text: bool = True,
    encoding: str = "utf-8",
    errors: str = "replace",
    metadata: Mapping[str, Any] | None = None,
    operator_session_id: str | None = None,
    context: Mapping[str, Any] | None = None,
    allow_paths: Any = None,
) -> dict[str, Any]:
    request = build_runtime_execution_request(
        command=command,
        shell=shell,
        cwd=cwd,
        timeout=timeout,
        input_text=input_text,
        env=env,
        capture_output=capture_output,
        text=text,
        encoding=encoding,
        errors=errors,
        metadata=metadata,
        operator_session_id=operator_session_id,
        context=context,
        allow_paths=allow_paths,
    )

    return execute_runtime_request(request).to_dict()
